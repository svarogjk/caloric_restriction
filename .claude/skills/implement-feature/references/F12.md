# F12 — Analysis History Dashboard

**Size**: M | **Requires**: F07 | **Priority**: 12

## Problem

Current conversation list shows text chats, not analysis history. Researchers need to revisit past analyses — a researcher with 50 stored analyses is not switching to KMPlot.

## Implementation

### Frontend: chatSlice.ts

```typescript
interface AnalysisHistorySummary {
  id: string
  query: string
  n_datasets: number
  n_genes: number
  created_at: string
}

// Add to ChatState:
analysisHistory: AnalysisHistorySummary[]
historyLoading: boolean

// Add thunk:
export const fetchAnalysisHistory = createAsyncThunk(
  'chat/fetchAnalysisHistory',
  async (_, { getState }) => {
    const state = getState() as RootState
    const token = state.auth.token
    if (!token) throw new Error('Not authenticated')
    return await getAnalysisHistory()  // from api.ts (F07)
  }
)

// In extraReducers:
builder
  .addCase(fetchAnalysisHistory.pending, (state) => { state.historyLoading = true })
  .addCase(fetchAnalysisHistory.fulfilled, (state, action) => {
    state.historyLoading = false
    state.analysisHistory = action.payload
  })
  .addCase(fetchAnalysisHistory.rejected, (state) => { state.historyLoading = false })
```

### Frontend: AnalysisHistoryPage.tsx

New `frontend/src/components/AnalysisHistoryPage.tsx`:
```tsx
export const AnalysisHistoryPage = () => {
  const dispatch = useAppDispatch()
  const { analysisHistory, historyLoading } = useAppSelector(s => s.chat)
  const navigate = (path: string) => { window.history.pushState({}, '', path); window.dispatchEvent(new Event('popstate')) }

  useEffect(() => { dispatch(fetchAnalysisHistory()) }, [dispatch])

  // Search filter
  const [search, setSearch] = useState('')
  const filtered = analysisHistory.filter(h =>
    h.query.toLowerCase().includes(search.toLowerCase())
  )

  const handleRerun = (item: AnalysisHistorySummary) => {
    // Pre-fill chat with the original query and navigate to main view
    dispatch(setAnalysisQuery(item.query))  // new action: sets chatInput
    navigate('/')
  }

  return (
    <div className="min-h-screen bg-gray-50 p-6">
      <div className="max-w-4xl mx-auto">
        <div className="flex items-center justify-between mb-6">
          <h1 className="text-2xl font-semibold text-gray-900">Analysis History</h1>
          <button onClick={() => navigate('/')} className="text-sm text-blue-600 hover:underline">
            ← Back to Chat
          </button>
        </div>

        <input type="text" placeholder="Search analyses..."
          value={search} onChange={e => setSearch(e.target.value)}
          className="w-full border rounded px-3 py-2 mb-4 text-sm" />

        {historyLoading ? (
          <div className="text-center py-8 text-gray-500">Loading history...</div>
        ) : filtered.length === 0 ? (
          <div className="text-center py-8 text-gray-400">No analyses found</div>
        ) : (
          <div className="bg-white rounded-lg border divide-y">
            {filtered.map(item => (
              <div key={item.id} className="flex items-center px-4 py-3 gap-4 hover:bg-gray-50">
                <div className="flex-1 min-w-0">
                  <p className="text-sm font-medium text-gray-900 truncate">{item.query}</p>
                  <p className="text-xs text-gray-500 mt-0.5">
                    {new Date(item.created_at).toLocaleDateString()} —
                    {item.n_datasets} datasets, {item.n_genes} genes
                  </p>
                </div>
                <div className="flex gap-2 shrink-0">
                  <button onClick={() => navigate(`/results/${item.id}`)}
                    className="text-xs px-2 py-1 border rounded hover:bg-gray-50 text-gray-600">
                    View
                  </button>
                  <button onClick={() => handleRerun(item)}
                    className="text-xs px-2 py-1 bg-blue-600 text-white rounded hover:bg-blue-700">
                    Re-run
                  </button>
                </div>
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  )
}
```
