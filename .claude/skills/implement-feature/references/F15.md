# F15 — Publication-Ready Export Package

**Size**: L | **Requires**: F07 | **Priority**: 15

## Problem

Researchers need paper-ready outputs. A tool that writes its own Methods section gets cited in papers — which is permanent, passive marketing. KMPlot's dominance is partly because its PNG output is directly paste-able into papers.

### Backend: Export Service

New `backend/app/services/export_service.py`:
```python
import io
import json
import zipfile
import math
from typing import Optional
import openpyxl
from openpyxl.styles import Font, PatternFill, Alignment
import matplotlib
matplotlib.use('Agg')  # non-interactive backend
import matplotlib.pyplot as plt
import matplotlib.patches as patches
import numpy as np

class ExportService:

    def build_export_zip(self, result_json: dict) -> bytes:
        """Build a ZIP containing Excel, SVG plots, and methods text."""
        buf = io.BytesIO()
        with zipfile.ZipFile(buf, 'w', zipfile.ZIP_DEFLATED) as zf:
            # 1. Gene table Excel
            zf.writestr('genes.xlsx', self._build_excel(result_json))
            # 2. Methods text
            zf.writestr('methods.txt', self._build_methods(result_json))
            # 3. Volcano plot SVG
            zf.writestr('volcano_plot.svg', self._build_volcano_svg(result_json))
            # 4. Forest plots for top 10 genes
            for gene in result_json.get('common_genes', [])[:10]:
                svg = self._build_forest_svg(gene)
                if svg:
                    safe_name = gene['gene_symbol'].replace('/', '_')
                    zf.writestr(f'forest_plots/{safe_name}.svg', svg)
        return buf.getvalue()

    def _build_excel(self, result_json: dict) -> bytes:
        wb = openpyxl.Workbook()
        ws = wb.active
        ws.title = "Survival Analysis Results"

        headers = ['Gene Symbol', 'Gene ID', 'N Datasets', 'Avg Hazard Ratio',
                   'Avg Cox P-Value', 'Direction Consistency %', 'Risk Direction', 'Dataset IDs']
        header_fill = PatternFill(start_color="366092", end_color="366092", fill_type="solid")
        header_font = Font(bold=True, color="FFFFFF")

        for col, header in enumerate(headers, 1):
            cell = ws.cell(row=1, column=col, value=header)
            cell.fill = header_fill
            cell.font = header_font

        for row, gene in enumerate(result_json.get('common_genes', []), 2):
            ws.cell(row=row, column=1, value=gene['gene_symbol'])
            ws.cell(row=row, column=2, value=gene['gene_id'])
            ws.cell(row=row, column=3, value=gene['n_datasets'])
            ws.cell(row=row, column=4, value=round(gene['avg_hazard_ratio'], 4))
            ws.cell(row=row, column=5, value=gene['avg_cox_p_value'])
            ws.cell(row=row, column=6, value=round(gene['risk_direction_consistency'] * 100, 1))
            ws.cell(row=row, column=7, value='High Risk' if gene['avg_hazard_ratio'] > 1 else 'Protective')
            ws.cell(row=row, column=8, value='; '.join(d['dataset_id'] for d in gene.get('datasets', [])))

        buf = io.BytesIO()
        wb.save(buf)
        return buf.getvalue()

    def _build_methods(self, result_json: dict) -> str:
        query = result_json.get('query', '')
        n_datasets = result_json.get('n_datasets_analyzed', 0)
        n_with_survival = result_json.get('n_datasets_with_survival', 0)
        n_genes = len(result_json.get('common_genes', []))
        model = result_json.get('model_used', 'AI-powered')
        min_occurrence = result_json.get('min_occurrence', 2)

        return f"""METHODS

Gene Expression Survival Analysis

Gene expression datasets were retrieved from the NCBI Gene Expression Omnibus (GEO;
https://www.ncbi.nlm.nih.gov/geo/) database using the search query "{query}".
Datasets were ranked by survival analysis potential using a large language model
({model})-based relevance scoring approach. Expression data were downloaded from
{n_datasets} datasets and processed using standard bioinformatics pipelines.
Probe-to-gene symbol mapping was performed using GEO platform annotation files.

For each dataset containing time-to-event clinical data, survival analysis was
performed as follows: samples were stratified into high- and low-expression groups
using the median expression value as threshold. Survival associations were identified
using Cox proportional hazards regression (lifelines library, Python). Genes were
considered significantly associated with survival at a two-sided p < 0.05 (log-rank
test) with |HR| > 1.5. Kaplan-Meier survival curves were generated using the lifelines
library.

Of {n_datasets} datasets searched, {n_with_survival} contained usable time-to-event
data. A total of {n_genes} genes were found to be consistently associated with survival,
defined as appearing in ≥ {min_occurrence} independent datasets with consistent
direction of effect.

Analyses were performed using GEO Survival Analysis
(https://github.com/your-repo/geo-survival-analysis).

CITATION
If using these results in publication, please cite:
[Your citation here]
"""

    def _build_volcano_svg(self, result_json: dict) -> str:
        genes = result_json.get('common_genes', [])
        if not genes:
            return '<svg><text>No data</text></svg>'

        fig, ax = plt.subplots(figsize=(8, 6))
        for gene in genes:
            hr = gene['avg_hazard_ratio']
            pval = gene['avg_cox_p_value']
            x = math.log2(hr) if hr > 0 else 0
            y = -math.log10(pval) if pval > 0 else 0
            color = 'red' if (hr > 1.5 and pval < 0.05) else 'steelblue' if (hr < 0.67 and pval < 0.05) else 'gray'
            ax.scatter(x, y, c=color, alpha=0.6, s=30)
            if gene['n_datasets'] >= 3 and pval < 0.01:
                ax.annotate(gene['gene_symbol'], (x, y), fontsize=6, alpha=0.7)

        ax.axhline(y=-math.log10(0.05), color='gray', linestyle='--', alpha=0.5, label='p=0.05')
        ax.axvline(x=0, color='gray', linestyle='-', alpha=0.3)
        ax.set_xlabel('log₂(Hazard Ratio)', fontsize=11)
        ax.set_ylabel('-log₁₀(p-value)', fontsize=11)
        ax.set_title(f"Survival Analysis: {result_json.get('query', '')[:50]}", fontsize=11)
        ax.legend()
        plt.tight_layout()

        buf = io.StringIO()
        fig.savefig(buf, format='svg')
        plt.close(fig)
        return buf.getvalue()

    def _build_forest_svg(self, gene: dict) -> Optional[str]:
        datasets = gene.get('datasets', [])
        valid = [d for d in datasets if d.get('hazard_ratio') and d.get('hr_ci_lower') and d.get('hr_ci_upper')]
        if not valid:
            return None

        fig, ax = plt.subplots(figsize=(7, max(3, len(valid) * 0.6 + 1.5)))
        y_positions = list(range(len(valid), 0, -1))

        for i, (dataset, y) in enumerate(zip(valid, y_positions)):
            hr = dataset['hazard_ratio']
            ci_lo = dataset['hr_ci_lower']
            ci_hi = dataset['hr_ci_upper']
            color = 'red' if hr > 1 else 'steelblue'
            ax.plot([ci_lo, ci_hi], [y, y], color=color, linewidth=1.5, alpha=0.7)
            ax.scatter(hr, y, marker='s', color=color, s=50, zorder=5)
            ax.text(-0.05, y, dataset.get('dataset_id', '')[:12], ha='right', va='center', fontsize=8)
            ax.text(ci_hi * 1.05, y, f"{hr:.2f} [{ci_lo:.2f}-{ci_hi:.2f}]", va='center', fontsize=7)

        ax.axvline(x=1, color='gray', linestyle='--', alpha=0.5)
        ax.set_yticks([])
        ax.set_xlabel('Hazard Ratio (95% CI)', fontsize=10)
        ax.set_title(f"Forest Plot: {gene['gene_symbol']}", fontsize=11)
        ax.set_xscale('log')
        plt.tight_layout()

        buf = io.StringIO()
        fig.savefig(buf, format='svg')
        plt.close(fig)
        return buf.getvalue()
```

### Backend: Endpoint

`backend/app/api/routes.py`:
```python
from fastapi.responses import StreamingResponse
from app.services.export_service import ExportService

@router.get("/api/analysis/{result_id}/export")
async def export_analysis(
    result_id: str,
    current_user = Depends(get_current_user)
):
    async with get_async_session() as session:
        result = await session.get(AnalysisResult, uuid_module.UUID(result_id))
        if not result:
            raise HTTPException(404, "Analysis not found")

    service = ExportService()
    zip_bytes = service.build_export_zip(result.result_json)
    safe_query = result.query.replace(' ', '_')[:30]

    return StreamingResponse(
        io.BytesIO(zip_bytes),
        media_type="application/zip",
        headers={"Content-Disposition": f"attachment; filename=geo_analysis_{safe_query}.zip"}
    )
```

### Frontend: Upgrade Export Button

```tsx
const downloadPackage = async () => {
  if (!savedAnalysisId) return
  const response = await fetch(`/api/analysis/${savedAnalysisId}/export`, {
    headers: { 'Authorization': `Bearer ${token}` }
  })
  const blob = await response.blob()
  const link = document.createElement('a')
  link.href = URL.createObjectURL(blob)
  link.download = `geo_analysis_${Date.now()}.zip`
  link.click()
}

<button onClick={downloadPackage}
  className="text-xs px-3 py-1.5 border rounded hover:bg-gray-50 text-gray-600">
  ↓ Download Publication Package
</button>
```
