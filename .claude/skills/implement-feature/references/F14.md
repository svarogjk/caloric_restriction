# F14 — Analysis Comparison Mode

**Size**: L | **Requires**: F07, F12 | **Priority**: 14

## Problem

No way to compare gene signatures across different cancer types or conditions. "Which genes are significant in both breast AND lung cancer?" is a core research question. This is the feature that turns a query tool into a research platform — and no competitor can offer it.

## Implementation

### Backend: Comparison Service

New `backend/app/services/comparison_service.py`:
```python
from dataclasses import dataclass
from typing import List, Dict, Optional
import numpy as np
from scipy.stats import spearmanr

@dataclass
class GeneComparisonData:
    gene_symbol: str
    gene_id: str
    hr_a: float
    hr_b: float
    p_value_a: float
    p_value_b: float
    datasets_a: int
    datasets_b: int

@dataclass
class ComparisonResult:
    query_a: str
    query_b: str
    shared_genes: List[GeneComparisonData]
    genes_only_in_a: List[str]   # gene symbols
    genes_only_in_b: List[str]
    n_total_a: int
    n_total_b: int
    hr_correlation: Optional[float]  # Spearman r on shared genes
    direction_agreement_rate: float  # % of shared genes with same risk direction

class ComparisonService:
    async def compare_analyses(
        self,
        result_json_a: dict,
        result_json_b: dict
    ) -> ComparisonResult:
        genes_a = {g['gene_symbol']: g for g in result_json_a.get('common_genes', [])}
        genes_b = {g['gene_symbol']: g for g in result_json_b.get('common_genes', [])}

        shared_symbols = set(genes_a.keys()) & set(genes_b.keys())
        only_in_a = sorted(set(genes_a.keys()) - shared_symbols)
        only_in_b = sorted(set(genes_b.keys()) - shared_symbols)

        shared_data = []
        for sym in sorted(shared_symbols):
            ga, gb = genes_a[sym], genes_b[sym]
            shared_data.append(GeneComparisonData(
                gene_symbol=sym, gene_id=ga['gene_id'],
                hr_a=ga['avg_hazard_ratio'], hr_b=gb['avg_hazard_ratio'],
                p_value_a=ga['avg_cox_p_value'], p_value_b=gb['avg_cox_p_value'],
                datasets_a=ga['n_datasets'], datasets_b=gb['n_datasets']
            ))

        hr_corr = None
        direction_agreement = 0.0
        if len(shared_data) >= 3:
            hrs_a = [g.hr_a for g in shared_data]
            hrs_b = [g.hr_b for g in shared_data]
            hr_corr = float(spearmanr(hrs_a, hrs_b).statistic)
            same_dir = sum(1 for g in shared_data if (g.hr_a > 1) == (g.hr_b > 1))
            direction_agreement = same_dir / len(shared_data)

        return ComparisonResult(
            query_a=result_json_a.get('query', ''),
            query_b=result_json_b.get('query', ''),
            shared_genes=shared_data,
            genes_only_in_a=only_in_a[:50],  # cap for response size
            genes_only_in_b=only_in_b[:50],
            n_total_a=len(genes_a),
            n_total_b=len(genes_b),
            hr_correlation=hr_corr,
            direction_agreement_rate=direction_agreement
        )
```

### Backend: Endpoint

`backend/app/api/routes.py`:
```python
from app.services.comparison_service import ComparisonService

@router.post("/api/analysis/compare")
async def compare_analyses(
    payload: dict,
    current_user = Depends(get_current_user)
):
    result_id_a = payload.get("result_id_a")
    result_id_b = payload.get("result_id_b")
    if not result_id_a or not result_id_b:
        raise HTTPException(400, "result_id_a and result_id_b are required")

    async with get_async_session() as session:
        ra = await session.get(AnalysisResult, uuid_module.UUID(result_id_a))
        rb = await session.get(AnalysisResult, uuid_module.UUID(result_id_b))
        if not ra or not rb:
            raise HTTPException(404, "One or both analyses not found")

    service = ComparisonService()
    result = await service.compare_analyses(ra.result_json, rb.result_json)
    return vars(result)
```

### Frontend: AnalysisHistoryPage.tsx

Add multi-select and Compare button:
```tsx
const [selectedIds, setSelectedIds] = useState<string[]>([])

const toggleSelect = (id: string) => {
  setSelectedIds(prev =>
    prev.includes(id) ? prev.filter(x => x !== id) : [...prev.slice(-1), id]  // max 2
  )
}

// In list rows, add checkbox:
<input type="checkbox" checked={selectedIds.includes(item.id)}
  onChange={() => toggleSelect(item.id)} className="mr-2" />

// Above list:
{selectedIds.length === 2 && (
  <button onClick={() => navigate(`/compare?a=${selectedIds[0]}&b=${selectedIds[1]}`)}
    className="px-4 py-2 bg-purple-600 text-white rounded text-sm hover:bg-purple-700">
    Compare Selected (2)
  </button>
)}
```
