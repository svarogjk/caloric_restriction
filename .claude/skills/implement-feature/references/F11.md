# F11 — Pathway / GO Enrichment Analysis

**Size**: M | **Priority**: 11

## Problem

A list of 50 survival genes has no biological context. Researchers currently copy gene lists into g:Profiler manually. Pathway enrichment transforms a gene list into a biological narrative ("enriched in cell cycle, DNA damage response"). High-value, low implementation barrier.

## Files to Read First

- `backend/app/api/routes.py` (pattern for adding new endpoints)
- `frontend/src/components/chat/AnalysisResultsDisplay.tsx` (Summary tab — where to add the enrichment button)
- `frontend/src/services/api.ts` (existing API call patterns)

## Implementation

### Backend: Enrichment Service

New file `backend/app/services/enrichment_service.py`:
```python
import httpx
from dataclasses import dataclass, field
from typing import List, Optional
import logging

logger = logging.getLogger(__name__)

@dataclass
class EnrichmentTerm:
    term_id: str
    term_name: str
    source: str      # "GO:BP", "GO:MF", "KEGG", "REAC"
    p_value: float
    fdr: float
    intersection_size: int
    query_size: int
    term_size: int
    intersection_genes: List[str] = field(default_factory=list)

    @property
    def external_url(self) -> str:
        if self.source == "KEGG":
            return f"https://www.genome.jp/dbget-bin/www_bget?{self.term_id}"
        if self.source == "REAC":
            return f"https://reactome.org/PathwayBrowser/#/{self.term_id.split(':')[1]}"
        # GO terms:
        return f"https://www.ebi.ac.uk/QuickGO/term/{self.term_id}"

class EnrichmentService:
    GPROFILER_URL = "https://biit.cs.ut.ee/gprofiler/api/gost/profile/"
    ORGANISM_MAP = {
        "Homo sapiens": "hsapiens",
        "Mus musculus": "mmusculus",
        "Rattus norvegicus": "rnorvegicus"
    }

    async def run_enrichment(
        self,
        genes: List[str],
        organism: str = "hsapiens"
    ) -> List[EnrichmentTerm]:
        org = self.ORGANISM_MAP.get(organism, organism)  # allow direct organism codes too
        payload = {
            "organism": org,
            "query": genes,
            "sources": ["GO:BP", "GO:MF", "KEGG", "REAC"],
            "significance_threshold_method": "fdr",
            "user_threshold": 0.05,
            "no_evidences": False,
            "ordered_query": False,
            "all_results": False
        }
        try:
            async with httpx.AsyncClient(timeout=30.0) as client:
                resp = await client.post(self.GPROFILER_URL, json=payload)
                resp.raise_for_status()
            data = resp.json()
        except httpx.TimeoutException:
            raise ValueError("Enrichment API timed out")
        except httpx.HTTPStatusError as e:
            raise ValueError(f"Enrichment API error: {e.response.status_code}")

        results = data.get("result", [])
        terms = []
        for r in results[:100]:  # cap at 100 terms
            intersections = r.get("intersections", [[]])
            genes_overlap = intersections[0] if intersections else []
            terms.append(EnrichmentTerm(
                term_id=r["native"],
                term_name=r["name"],
                source=r["source"],
                p_value=r["p_value"],
                fdr=r["p_value"],  # g:Profiler returns BH-corrected p-value directly
                intersection_size=r["intersection_size"],
                query_size=r["query_size"],
                term_size=r["term_size"],
                intersection_genes=genes_overlap
            ))
        return sorted(terms, key=lambda t: t.fdr)
```

### Backend: Endpoint

`backend/app/api/routes.py`:
```python
from app.services.enrichment_service import EnrichmentService

@router.post("/api/enrichment")
async def run_enrichment(payload: dict):
    genes = payload.get("genes", [])
    organism = payload.get("organism", "hsapiens")
    if not genes:
        raise HTTPException(400, "genes list is required")
    if len(genes) > 500:
        raise HTTPException(400, "genes list must contain at most 500 symbols")
    service = EnrichmentService()
    try:
        terms = await service.run_enrichment(genes, organism)
    except ValueError as e:
        raise HTTPException(502, str(e))
    return {"terms": [vars(t) for t in terms], "query_size": len(genes)}
```

### Frontend: EnrichmentResults Component

New `frontend/src/components/chat/EnrichmentResults.tsx`:
```tsx
interface EnrichmentTerm {
  term_id: string; term_name: string; source: string
  p_value: number; fdr: number; intersection_size: number
  query_size: number; term_size: number
  intersection_genes: string[]; external_url: string
}

interface Props { terms: EnrichmentTerm[]; querySize: number }

export const EnrichmentResults = ({ terms, querySize }: Props) => {
  const [sourceFilter, setSourceFilter] = useState<string>('ALL')
  const sources = ['ALL', 'GO:BP', 'GO:MF', 'KEGG', 'REAC']
  const filtered = terms.filter(t => sourceFilter === 'ALL' || t.source === sourceFilter)

  return (
    <div className="mt-4 border rounded-lg overflow-hidden">
      <div className="bg-gray-50 px-3 py-2 flex items-center gap-2 border-b">
        <span className="text-xs font-medium text-gray-700">Pathway Enrichment</span>
        <span className="text-xs text-gray-400">({filtered.length} significant terms, q<0.05)</span>
        <div className="ml-auto flex gap-1">
          {sources.map(s => (
            <button key={s} onClick={() => setSourceFilter(s)}
              className={`text-xs px-2 py-0.5 rounded ${sourceFilter === s ? 'bg-blue-600 text-white' : 'border hover:bg-gray-100'}`}>
              {s}
            </button>
          ))}
        </div>
      </div>
      <div className="max-h-64 overflow-y-auto">
        {filtered.slice(0, 50).map(t => (
          <div key={t.term_id} className="flex items-center px-3 py-1.5 border-b hover:bg-gray-50 text-xs">
            <span className="w-16 text-gray-400 shrink-0">{t.source}</span>
            <a href={t.external_url} target="_blank" rel="noopener" className="flex-1 text-blue-600 hover:underline truncate">
              {t.term_name}
            </a>
            <span className="w-20 text-right text-gray-500">{t.intersection_size}/{t.term_size} genes</span>
            <span className="w-24 text-right text-gray-600 font-mono">q={t.fdr.toExponential(2)}</span>
          </div>
        ))}
      </div>
    </div>
  )
}
```

### Frontend: AnalysisResultsDisplay.tsx

In the Summary tab, add below the top-genes section:
```tsx
const [enrichmentResults, setEnrichmentResults] = useState<EnrichmentTerm[] | null>(null)
const [enrichmentLoading, setEnrichmentLoading] = useState(false)

const runEnrichment = async () => {
  if (!results?.common_genes) return
  // Get significant genes (filter by p-value threshold)
  const sigGenes = results.common_genes
    .filter(g => g.avg_cox_p_value < 0.05)
    .map(g => g.gene_symbol)
  if (sigGenes.length === 0) return

  setEnrichmentLoading(true)
  try {
    const resp = await axios.post('/api/enrichment',
      { genes: sigGenes, organism: 'hsapiens' },
      { headers: authHeaders() }
    )
    setEnrichmentResults(resp.data.terms)
  } catch (e) {
    console.error('Enrichment failed:', e)
  } finally {
    setEnrichmentLoading(false)
  }
}

// Add after top genes preview:
<button onClick={runEnrichment} disabled={enrichmentLoading}
  className="mt-3 w-full text-sm py-2 border border-purple-300 text-purple-700 rounded hover:bg-purple-50 disabled:opacity-50">
  {enrichmentLoading ? 'Running pathway enrichment...' : 'Run Pathway Enrichment Analysis'}
</button>
{enrichmentResults && <EnrichmentResults terms={enrichmentResults} querySize={/* sig gene count */} />}
```
