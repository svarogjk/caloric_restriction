# F09 — Shareable Result URLs

**Size**: M | **Requires**: F07 | **Priority**: 9

## Problem

No viral growth mechanism. Researchers can't share results with colleagues. A shareable URL is how a bioinformatician convinces their PI, and how a PI discovers the tool from a colleague's link.

## Implementation

### Backend: Public Endpoint (no auth required)

`backend/app/api/routes.py` — Add (no Depends(get_current_user)):
```python
@router.get("/api/analysis/{result_id}/public")
async def get_public_analysis(result_id: str):
    """Return analysis result without authentication (read-only public access)."""
    try:
        result_uuid = uuid_module.UUID(result_id)
    except ValueError:
        raise HTTPException(status_code=400, detail="Invalid result ID format")

    async with get_async_session() as session:
        result = await session.get(AnalysisResult, result_uuid)
        if not result:
            raise HTTPException(status_code=404, detail="Analysis result not found")
        return result.result_json
```

### Frontend: Public Result View

New `frontend/src/components/PublicResultView.tsx`:
```tsx
interface Props { resultId: string }

export const PublicResultView = ({ resultId }: Props) => {
  const [result, setResult] = useState<AnalysisResult | null>(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    fetch(`/api/analysis/${resultId}/public`)
      .then(r => r.ok ? r.json() : Promise.reject(r.statusText))
      .then(setResult)
      .catch(e => setError(String(e)))
      .finally(() => setLoading(false))
  }, [resultId])

  if (loading) return <div className="flex items-center justify-center h-screen">Loading...</div>
  if (error) return <div className="text-center p-8 text-red-600">Result not found or unavailable</div>

  return (
    <div className="min-h-screen bg-gray-50">
      {/* Sign-up CTA banner */}
      <div className="bg-blue-600 text-white text-center py-2 px-4 text-sm">
        Shared via GEO Survival Analysis —
        <button onClick={() => window.location.href = '/'} className="ml-1 underline">
          Sign up to run your own analyses
        </button>
      </div>
      <div className="max-w-5xl mx-auto p-6">
        <h1 className="text-xl font-semibold mb-4">Shared Analysis: {result?.query}</h1>
        <AnalysisResultsDisplay results={result} isPublicView={true} />
      </div>
    </div>
  )
}
```

### Frontend: Share Button in AnalysisResultsDisplay.tsx

```tsx
// Props addition (if not isPublicView):
const [shareCopied, setShareCopied] = useState(false)

const handleShare = async () => {
  if (!savedAnalysisId) return
  const url = `${window.location.origin}/results/${savedAnalysisId}`
  await navigator.clipboard.writeText(url)
  setShareCopied(true)
  setTimeout(() => setShareCopied(false), 2500)
}

// Add button in export toolbar (from F04) or near title:
{savedAnalysisId && !isPublicView && (
  <button onClick={handleShare}
    className="text-xs px-3 py-1.5 border border-gray-300 rounded hover:bg-gray-50 text-gray-600">
    {shareCopied ? '✓ Link Copied!' : '🔗 Share'}
  </button>
)}
```
